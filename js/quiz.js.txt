let currentQuizData = null;
let currentQuestionIndex = 0;
let userAnswers = [];
let quizStartTime = null;

// INICIAR QUIZ
function iniciarQuiz(perguntas) {
    console.log('Iniciando quiz com', perguntas.length, 'perguntas');
    
    currentQuizData = perguntas;
    currentQuestionIndex = 0;
    userAnswers = [];
    quizStartTime = Date.now();
    
    // Limpar resultado anterior
    const resultadoDiv = document.getElementById('quiz-resultado');
    if (resultadoDiv) {
        resultadoDiv.classList.add('hidden');
    }
    
    // Renderizar primeira pergunta
    renderizarPergunta();
}

// RENDERIZAR PERGUNTA ATUAL
function renderizarPergunta() {
    const container = document.getElementById('quiz-container');
    
    if (!container || !currentQuizData) {
        console.error('Container do quiz não encontrado ou dados inválidos');
        return;
    }
    
    const pergunta = currentQuizData[currentQuestionIndex];
    const totalPerguntas = currentQuizData.length;
    
    container.innerHTML = `
        <div class="quiz-progress mb-6">
            <div class="flex justify-between items-center mb-2">
                <span class="text-sm font-medium text-gray-600">Pergunta ${currentQuestionIndex + 1} de ${totalPerguntas}</span>
                <span class="text-sm font-medium text-primary">${Math.round(((currentQuestionIndex + 1) / totalPerguntas) * 100)}%</span>
            </div>
            <div class="w-full h-2 bg-gray-200 rounded-full overflow-hidden">
                <div class="h-full bg-primary transition-all duration-300" style="width: ${((currentQuestionIndex + 1) / totalPerguntas) * 100}%"></div>
            </div>
        </div>
        
        <div class="quiz-question">
            <h3 class="text-xl font-bold text-gray-900 mb-6">${pergunta.pergunta}</h3>
            
            <div class="quiz-options space-y-3">
                ${pergunta.opcoes.map((opcao, index) => `
                    <label class="quiz-option cursor-pointer" data-index="${index}">
                        <input type="radio" name="resposta" value="${index}" class="hidden">
                        <div class="option-content flex items-center gap-3 p-4 border-2 border-gray-300 rounded-lg transition-all hover:border-primary hover:bg-blue-50">
                            <div class="option-letter w-8 h-8 rounded-full bg-gray-100 flex items-center justify-center font-semibold text-gray-700">
                                ${String.fromCharCode(65 + index)}
                            </div>
                            <span class="option-text flex-1">${opcao}</span>
                        </div>
                    </label>
                `).join('')}
            </div>
            
            <div class="mt-8 flex gap-4">
                ${currentQuestionIndex > 0 ? `
                    <button onclick="voltarPergunta()" class="btn-secondary px-6 py-3 rounded-lg font-semibold">
                        Voltar
                    </button>
                ` : ''}
                <button id="btn-proximo-quiz" onclick="proximaPergunta()" disabled class="flex-1 btn-primary px-6 py-3 rounded-lg font-semibold opacity-50 cursor-not-allowed">
                    ${currentQuestionIndex < totalPerguntas - 1 ? 'Próxima Pergunta' : 'Finalizar Quiz'}
                </button>
            </div>
        </div>
    `;
    
    // Adicionar event listeners nas opções
    const opcoes = container.querySelectorAll('.quiz-option');
    opcoes.forEach(opcao => {
        opcao.addEventListener('click', function() {
            selecionarOpcao(this);
        });
    });
    
    // Se já existe resposta para esta pergunta, marcar
    if (userAnswers[currentQuestionIndex] !== undefined) {
        const respostaSalva = userAnswers[currentQuestionIndex];
        const opcaoSalva = container.querySelector(`.quiz-option[data-index="${respostaSalva}"]`);
        if (opcaoSalva) {
            selecionarOpcao(opcaoSalva);
        }
    }
}

// SELECIONAR OPÇÃO
function selecionarOpcao(element) {
    const container = document.getElementById('quiz-container');
    
    // Remover seleção de todas as opções
    const todasOpcoes = container.querySelectorAll('.quiz-option');
    todasOpcoes.forEach(opt => {
        opt.classList.remove('selected');
        const content = opt.querySelector('.option-content');
        content.classList.remove('border-primary', 'bg-blue-50');
        content.classList.add('border-gray-300');
    });
    
    // Marcar opção selecionada
    element.classList.add('selected');
    const content = element.querySelector('.option-content');
    content.classList.remove('border-gray-300');
    content.classList.add('border-primary', 'bg-blue-50');
    
    // Marcar radio input
    const radio = element.querySelector('input[type="radio"]');
    if (radio) {
        radio.checked = true;
    }
    
    // Habilitar botão próximo
    const btnProximo = document.getElementById('btn-proximo-quiz');
    if (btnProximo) {
        btnProximo.disabled = false;
        btnProximo.classList.remove('opacity-50', 'cursor-not-allowed');
    }
    
    // Salvar resposta
    const index = parseInt(element.getAttribute('data-index'));
    userAnswers[currentQuestionIndex] = index;
}

// VOLTAR PERGUNTA
function voltarPergunta() {
    if (currentQuestionIndex > 0) {
        currentQuestionIndex--;
        renderizarPergunta();
    }
}

// PRÓXIMA PERGUNTA
function proximaPergunta() {
    // Verificar se respondeu
    if (userAnswers[currentQuestionIndex] === undefined) {
        showToast('Por favor, selecione uma resposta', 'warning');
        return;
    }
    
    // Se não for a última pergunta, avançar
    if (currentQuestionIndex < currentQuizData.length - 1) {
        currentQuestionIndex++;
        renderizarPergunta();
    } else {
        // Última pergunta - finalizar quiz
        finalizarQuiz();
    }
}

// FINALIZAR QUIZ
function finalizarQuiz() {
    console.log('Finalizando quiz...');
    
    // Calcular resultado
    let acertos = 0;
    const totalPerguntas = currentQuizData.length;
    
    currentQuizData.forEach((pergunta, index) => {
        if (userAnswers[index] === pergunta.correta) {
            acertos++;
        }
    });
    
    const percentualAcerto = Math.round((acertos / totalPerguntas) * 100);
    const aprovado = percentualAcerto >= 70; // 70% para aprovação
    
    const tempoTotal = Math.round((Date.now() - quizStartTime) / 1000); // em segundos
    const tempoFormatado = formatarTempo(tempoTotal);
    
    // Salvar resultado
    salvarResultadoQuiz(acertos, totalPerguntas, percentualAcerto, tempoTotal);
    
    // Mostrar resultado
    mostrarResultado(acertos, totalPerguntas, percentualAcerto, aprovado, tempoFormatado);
    
    // Track evento
    trackQuizEvent('quiz_completed', {
        acertos: acertos,
        total: totalPerguntas,
        percentual: percentualAcerto,
        aprovado: aprovado,
        tempo: tempoTotal
    });
}

// MOSTRAR RESULTADO
function mostrarResultado(acertos, total, percentual, aprovado, tempo) {
    const container = document.getElementById('quiz-container');
    const resultadoDiv = document.getElementById('quiz-resultado');
    
    if (!resultadoDiv) {
        console.error('Div de resultado não encontrada');
        return;
    }
    
    // Ocultar perguntas
    if (container) {
        container.innerHTML = '';
    }
    
    // Mensagem baseada no desempenho
    let mensagem = '';
    let emoji = '';
    
    if (percentual >= 90) {
        mensagem = 'Excelente! Você dominou o conteúdo!';
        emoji = '🎉';
    } else if (percentual >= 70) {
        mensagem = 'Parabéns! Você foi aprovado!';
        emoji = '✅';
    } else if (percentual >= 50) {
        mensagem = 'Quase lá! Revise o conteúdo e tente novamente.';
        emoji = '📚';
    } else {
        mensagem = 'Recomendamos revisar o conteúdo antes de prosseguir.';
        emoji = '💪';
    }
    
    // Montar HTML do resultado
    resultadoDiv.innerHTML = `
        <div class="text-center">
            <div class="text-6xl mb-4">${emoji}</div>
            <h3 class="text-2xl font-bold mb-2 ${aprovado ? 'text-green-600' : 'text-orange-600'}">
                ${aprovado ? 'Aprovado!' : 'Não Aprovado'}
            </h3>
            <p class="text-lg text-gray-700 mb-6">${mensagem}</p>
            
            <div class="bg-gray-50 rounded-lg p-6 mb-6">
                <div class="grid grid-cols-3 gap-4 text-center">
                    <div>
                        <div class="text-3xl font-bold text-primary">${acertos}</div>
                        <div class="text-sm text-gray-600">Acertos</div>
                    </div>
                    <div>
                        <div class="text-3xl font-bold ${percentual >= 70 ? 'text-green-600' : 'text-orange-600'}">${percentual}%</div>
                        <div class="text-sm text-gray-600">Aproveitamento</div>
                    </div>
                    <div>
                        <div class="text-3xl font-bold text-gray-700">${tempo}</div>
                        <div class="text-sm text-gray-600">Tempo</div>
                    </div>
                </div>
            </div>
            
            <div class="space-y-3">
                ${aprovado ? `
                    <button onclick="proximaAula()" class="w-full btn-primary py-3 rounded-lg font-semibold">
                        Continuar para Próxima Aula
                    </button>
                ` : `
                    <button onclick="refazerQuiz()" class="w-full btn-primary py-3 rounded-lg font-semibold">
                        Refazer Quiz
                    </button>
                `}
                <button onclick="verGabarito()" class="w-full btn-secondary py-3 rounded-lg font-semibold">
                    Ver Gabarito
                </button>
            </div>
        </div>
    `;
    
    resultadoDiv.classList.remove('hidden');
    
    // Se aprovado, marcar aula como concluída automaticamente
    if (aprovado && typeof marcarComoConcluida === 'function') {
        const checkbox = document.getElementById('checkbox-concluida');
        if (checkbox && !checkbox.checked) {
            checkbox.checked = true;
            marcarComoConcluida();
        }
    }
}

// REFAZER QUIZ
function refazerQuiz() {
    const resultadoDiv = document.getElementById('quiz-resultado');
    if (resultadoDiv) {
        resultadoDiv.classList.add('hidden');
    }
    
    // Reiniciar quiz
    iniciarQuiz(currentQuizData);
}

// VER GABARITO
function verGabarito() {
    const container = document.getElementById('quiz-container');
    
    if (!container || !currentQuizData) {
        return;
    }
    
    let html = '<div class="gabarito-container space-y-6">';
    html += '<h3 class="text-2xl font-bold text-gray-900 mb-6">Gabarito Detalhado</h3>';
    
    currentQuizData.forEach((pergunta, index) => {
        const respostaUsuario = userAnswers[index];
        const respostaCorreta = pergunta.correta;
        const acertou = respostaUsuario === respostaCorreta;
        
        html += `
            <div class="gabarito-item border-2 ${acertou ? 'border-green-200 bg-green-50' : 'border-red-200 bg-red-50'} rounded-lg p-6">
                <div class="flex items-start gap-3 mb-4">
                    <div class="w-8 h-8 rounded-full ${acertou ? 'bg-green-500' : 'bg-red-500'} flex items-center justify-center text-white font-bold flex-shrink-0">
                        ${index + 1}
                    </div>
                    <div class="flex-1">
                        <p class="font-semibold text-gray-900 mb-3">${pergunta.pergunta}</p>
                        
                        <div class="space-y-2">
                            ${pergunta.opcoes.map((opcao, opcaoIndex) => {
                                const isCorreta = opcaoIndex === respostaCorreta;
                                const isUsuario = opcaoIndex === respostaUsuario;
                                
                                let classes = 'p-3 rounded-lg border-2 ';
                                if (isCorreta) {
                                    classes += 'border-green-500 bg-green-100';
                                } else if (isUsuario && !isCorreta) {
                                    classes += 'border-red-500 bg-red-100';
                                } else {
                                    classes += 'border-gray-200 bg-white';
                                }
                                
                                return `
                                    <div class="${classes}">
                                        <div class="flex items-center gap-2">
                                            <span class="font-semibold">${String.fromCharCode(65 + opcaoIndex)}.</span>
                                            <span>${opcao}</span>
                                            ${isCorreta ? '<span class="ml-auto text-green-600 font-semibold">✓ Correta</span>' : ''}
                                            ${isUsuario && !isCorreta ? '<span class="ml-auto text-red-600 font-semibold">✗ Sua resposta</span>' : ''}
                                        </div>
                                    </div>
                                `;
                            }).join('')}
                        </div>
                    </div>
                </div>
            </div>
        `;
    });
    
    html += `
        <div class="flex gap-4 mt-8">
            <button onclick="refazerQuiz()" class="flex-1 btn-primary py-3 rounded-lg font-semibold">
                Refazer Quiz
            </button>
            <button onclick="proximaAula()" class="flex-1 btn-secondary py-3 rounded-lg font-semibold">
                Próxima Aula
            </button>
        </div>
    `;
    html += '</div>';
    
    container.innerHTML = html;
    
    // Ocultar resultado
    const resultadoDiv = document.getElementById('quiz-resultado');
    if (resultadoDiv) {
        resultadoDiv.classList.add('hidden');
    }
}

// SALVAR RESULTADO DO QUIZ
function salvarResultadoQuiz(acertos, total, percentual, tempo) {
    const urlParams = new URLSearchParams(window.location.search);
    const cursoId = urlParams.get('cursoId');
    const moduloNum = urlParams.get('modulo');
    const aulaNum = urlParams.get('aula');
    
    if (!cursoId || !moduloNum || !aulaNum) {
        console.error('Parâmetros inválidos para salvar quiz');
        return;
    }
    
    const key = `quiz_result_${cursoId}_${moduloNum}_${aulaNum}`;
    
    const resultado = {
        acertos: acertos,
        total: total,
        percentual: percentual,
        tempo: tempo,
        aprovado: percentual >= 70,
        data: new Date().toISOString(),
        tentativas: obterNumeroTentativas(key) + 1
    };
    
    try {
        localStorage.setItem(key, JSON.stringify(resultado));
        console.log('Resultado do quiz salvo:', resultado);
    } catch (e) {
        console.error('Erro ao salvar resultado do quiz:', e);
    }
}

// OBTER NÚMERO DE TENTATIVAS
function obterNumeroTentativas(key) {
    try {
        const stored = localStorage.getItem(key);
        if (stored) {
            const data = JSON.parse(stored);
            return data.tentativas || 0;
        }
    } catch (e) {
        console.error('Erro ao obter tentativas:', e);
    }
    return 0;
}

// OBTER RESULTADO DO QUIZ
function obterResultadoQuiz(cursoId, moduloNum, aulaNum) {
    const key = `quiz_result_${cursoId}_${moduloNum}_${aulaNum}`;
    
    try {
        const stored = localStorage.getItem(key);
        if (stored) {
            return JSON.parse(stored);
        }
    } catch (e) {
        console.error('Erro ao obter resultado do quiz:', e);
    }
    
    return null;
}

// FORMATAR TEMPO
function formatarTempo(segundos) {
    const minutos = Math.floor(segundos / 60);
    const segs = segundos % 60;
    
    if (minutos > 0) {
        return `${minutos}min ${segs}s`;
    } else {
        return `${segs}s`;
    }
}

// TRACK EVENTO DO QUIZ
function trackQuizEvent(eventType, eventData = {}) {
    const urlParams = new URLSearchParams(window.location.search);
    const cursoId = urlParams.get('cursoId');
    const moduloNum = urlParams.get('modulo');
    const aulaNum = urlParams.get('aula');
    
    const event = {
        type: eventType,
        cursoId: cursoId,
        modulo: moduloNum,
        aula: aulaNum,
        timestamp: new Date().toISOString(),
        ...eventData
    };
    
    console.log('Quiz Event:', event);
    
    // Aqui você pode enviar para analytics
}

// RESETAR PROGRESSO DO QUIZ (Para debug/admin)
function resetarQuiz(cursoId, moduloNum, aulaNum) {
    const key = `quiz_result_${cursoId}_${moduloNum}_${aulaNum}`;
    
    try {
        localStorage.removeItem(key);
        console.log('Quiz resetado:', key);
        return true;
    } catch (e) {
        console.error('Erro ao resetar quiz:', e);
        return false;
    }
}

// ESTATÍSTICAS GERAIS DE QUIZZES
function obterEstatisticasQuizzes(cursoId) {
    const stats = {
        totalQuizzes: 0,
        quizzesAprovados: 0,
        mediaAcertos: 0,
        tempoTotal: 0
    };
    
    try {
        for (let i = 0; i < localStorage.length; i++) {
            const key = localStorage.key(i);
            
            if (key.startsWith(`quiz_result_${cursoId}_`)) {
                const data = JSON.parse(localStorage.getItem(key));
                
                stats.totalQuizzes++;
                if (data.aprovado) {
                    stats.quizzesAprovados++;
                }
                stats.mediaAcertos += data.percentual;
                stats.tempoTotal += data.tempo;
            }
        }
        
        if (stats.totalQuizzes > 0) {
            stats.mediaAcertos = Math.round(stats.mediaAcertos / stats.totalQuizzes);
        }
        
    } catch (e) {
        console.error('Erro ao calcular estatísticas:', e);
    }
    
    return stats;
}
